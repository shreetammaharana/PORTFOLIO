  <script>
    const PASSWORD_KEY          = 'site_password';
    const PASSWORD_SET_ONCE_KEY = 'site_password_set_once';
    const CONTACTS_KEY          = 'contacts_secure';
    const ITEMS_KEY             = 'cards';
    const DAILY_NOTES_KEY       = 'notes_daily';
    const PERSONAL_NOTES_KEY    = 'notes_personal';

    const photosGrid = document.getElementById('photosGrid');
    const pdfGrid    = document.getElementById('pdfGrid');
    const extraGrid  = document.getElementById('extraGrid');

    function getPassword(){
      const setOnce = localStorage.getItem(PASSWORD_SET_ONCE_KEY) === 'true';
      if(!setOnce) return null;
      const saved = localStorage.getItem(PASSWORD_KEY);
      return saved && saved.trim() ? saved : null;
    }

    function setPassword(){
      const alreadySet = localStorage.getItem(PASSWORD_SET_ONCE_KEY) === 'true';
      if(alreadySet){
        alert('Password is already set and cannot be changed.');
        return;
      }
      const pwd = prompt('Set new password (only once):','');
      if(pwd === null) return;
      const trimmed = pwd.trim();
      if(!trimmed){
        alert('Password cannot be empty.');
        return;
      }
      localStorage.setItem(PASSWORD_KEY, trimmed);
      localStorage.setItem(PASSWORD_SET_ONCE_KEY, 'true');
      alert('Password set successfully. It cannot be changed again.');
      lockAllSecure();
    }

    function forgetPassword(){
      lockAllSecure();
      alert(
        'Warning: Password reset is disabled for security.\n' +
        'You cannot access hidden folders or personal notes using "Forgot password".'
      );
    }

    const navToggle = document.getElementById('navToggle');
    const navLinks  = document.getElementById('navLinks');
    navToggle.addEventListener('click', () => {
      navLinks.classList.toggle('open');
    });

    const pages = {
      home: document.getElementById('page-home'),
      note: document.getElementById('page-note'),
      items: document.getElementById('page-items'),
      services: document.getElementById('page-services'),
      contact: document.getElementById('page-contact')
    };

    function showPage(name){
      lockAllSecure();
      Object.keys(pages).forEach(key=>{
        pages[key].classList.toggle('active', key === name);
      });
      document.querySelectorAll('.nav-links a').forEach(a=>{
        const target = a.getAttribute('data-page');
        a.classList.toggle('active', target === name);
      });
      navLinks.classList.remove('open');
      window.scrollTo({top:0, behavior:'smooth'});
    }

    document.querySelectorAll('.nav-links a').forEach(a=>{
      a.addEventListener('click', (e)=>{
        e.preventDefault();
        const page = a.getAttribute('data-page');
        if(page) showPage(page);
      });
    });

    function savePersonal(e){
      e.preventDefault();
      const data = {
        name: document.getElementById('name').value,
        email: document.getElementById('email').value,
        bio: document.getElementById('bio').value
      };
      localStorage.setItem('personal', JSON.stringify(data));
      alert('Personal info saved safely in this browser.');
    }

    function loadPersonal(){
      const data = JSON.parse(localStorage.getItem('personal') || 'null');
      if(!data) return;
      document.getElementById('name').value = data.name || '';
      document.getElementById('email').value = data.email || '';
      document.getElementById('bio').value   = data.bio || '';
    }

    function loadSample(){
      document.getElementById('name').value   = 'Demo User';
      document.getElementById('email').value  = 'demo@example.com';
      document.getElementById('bio').value    = 'I like building cool web projects.';
    }

    let itemsFolderOpen      = false;
    let itemsUnlocked        = false;
    let contactsUnlocked     = false;
    let personalNotesUnlocked= false;

    function lockItems(){
      itemsUnlocked   = false;
      itemsFolderOpen = false;
      document.getElementById('itemsFolderContent').style.display = 'none';
      document.getElementById('unlockItemsBtn').textContent = '🔒 Unlock items';
    }

    function lockContacts(){
      contactsUnlocked = false;
      const list = document.getElementById('contactList');
      if(list){
        list.style.display = 'none';
        list.textContent = 'No contacts saved yet.';
      }
      document.getElementById('unlockBtn').textContent = '🔒 Unlock contacts';
    }

    function lockPersonalNotes(){
      personalNotesUnlocked = false;
      const sel = document.getElementById('noteTypeSelect');
      if(sel && sel.value === 'personal'){
        sel.value = 'daily';
      }
      renderNotes();
    }

    function lockAllSecure(){
      lockItems();
      lockContacts();
      lockPersonalNotes();
    }

    function toggleItemsFolder(){
      if(!itemsUnlocked){
        alert('Unlock items first.');
        return;
      }
      itemsFolderOpen = !itemsFolderOpen;
      document.getElementById('itemsFolderContent').style.display =
        itemsFolderOpen ? 'block' : 'none';
    }

    function unlockItems(){
      lockContacts();

      const pwdStored = getPassword();
      if(!pwdStored){
        alert('Password is not set yet. Owner must set the password from Personal info.');
        return;
      }

      const pwd = prompt('Enter password to view items:');
      if(pwd === null) return;
      if(pwd === pwdStored){
        itemsUnlocked   = true;
        itemsFolderOpen = true;
        document.getElementById('itemsFolderContent').style.display = 'block';
        document.getElementById('unlockItemsBtn').textContent = '🔓 Items unlocked';
      } else {
        alert('Wrong password.');
      }
    }

    function goToMyItems(){
      showPage('items');
    }

    function handleImage(e){
      const f = e.target.files[0];
      if(!f) return;
      const reader = new FileReader();
      reader.onload = function(ev){
        const dataURL = ev.target.result;
        addCard({type:'image', title:f.name, src:dataURL});
        goToMyItems();
      };
      reader.readAsDataURL(f);
    }

    function handleDoc(e){
      const f = e.target.files[0];
      if(!f) return;
      let docs = JSON.parse(localStorage.getItem('docs') || '[]');
      docs.push({name:f.name, size:f.size, uploaded:new Date().toISOString()});
      localStorage.setItem('docs', JSON.stringify(docs));
      addCard({type:'doc', title:f.name, src:null});
      goToMyItems();
    }

    function addCard(obj){
      const arr = JSON.parse(localStorage.getItem(ITEMS_KEY) || '[]');
      arr.unshift(obj);
      localStorage.setItem(ITEMS_KEY, JSON.stringify(arr));
      renderCards();
    }

    function renderCards(){
      const arr = JSON.parse(localStorage.getItem(ITEMS_KEY) || '[]');

      photosGrid.innerHTML = '';
      pdfGrid.innerHTML    = '';
      extraGrid.innerHTML  = '';

      if(!arr.length){
        photosGrid.innerHTML = '<div class="muted-note">No photos yet.</div>';
        pdfGrid.innerHTML    = '<div class="muted-note">No PDFs yet.</div>';
        extraGrid.innerHTML  = '<div class="muted-note">No extra files yet.</div>';
        return;
      }

      arr.forEach((c,i)=>{
        const div = document.createElement('div');
        div.className = 'card';

        if(c.type === 'image'){
          const img = document.createElement('img');
          img.src = c.src;
          div.appendChild(img);
        }else if(c.type === 'doc'){
          const docBox = document.createElement('div');
          docBox.style.padding = '10px';
          docBox.style.border = '1px dashed #d1d5db';
          docBox.style.borderRadius = '8px';
          docBox.textContent = 'Document: ' + (c.title || 'Untitled');
          div.appendChild(docBox);
        }else{
          const placeholder = document.createElement('div');
          placeholder.style.height='120px';
          placeholder.style.background='#eef2ff';
          placeholder.style.borderRadius='8px';
          div.appendChild(placeholder);
        }

        const title = document.createElement('div');
        title.className = 'title';
        title.textContent = c.title || 'Untitled';
        div.appendChild(title);

        const remove = document.createElement('button');
        remove.className = 'btn';
        remove.textContent = 'Remove';
        remove.onclick = ()=>{ removeCard(i); };
        div.appendChild(remove);

        let targetGrid = extraGrid;
        const nameLower = (c.title || '').toLowerCase();

        if(c.type === 'image'){
          targetGrid = photosGrid;
        } else if(c.type === 'doc' && nameLower.endsWith('.pdf')){
          targetGrid = pdfGrid;
        } else {
          targetGrid = extraGrid;
        }

        targetGrid.appendChild(div);
      });

      if(!photosGrid.innerHTML.trim()){
        photosGrid.innerHTML = '<div class="muted-note">No photos yet.</div>';
      }
      if(!pdfGrid.innerHTML.trim()){
        pdfGrid.innerHTML = '<div class="muted-note">No PDFs yet.</div>';
      }
      if(!extraGrid.innerHTML.trim()){
        extraGrid.innerHTML = '<div class="muted-note">No extra files yet.</div>';
      }
    }

    function removeCard(index){
      const arr = JSON.parse(localStorage.getItem(ITEMS_KEY) || '[]');
      arr.splice(index,1);
      localStorage.setItem(ITEMS_KEY, JSON.stringify(arr));
      renderCards();
    }

        localStorage.removeItem(ITEMS_KEY);

    function doSearch(e){
      e.preventDefault();
      const q = document.getElementById('searchInput').value.toLowerCase().trim();
      const arr = JSON.parse(localStorage.getItem(ITEMS_KEY) || '[]');

      showPage('items');
      if(!itemsUnlocked){
        alert('Unlock items first to search.');
        return;
      }
      if(!itemsFolderOpen){
        itemsFolderOpen = true;
        document.getElementById('itemsFolderContent').style.display = 'block';
      }

      if(!q){
        renderCards();
        alert('Search cleared. Showing all items in folders.');
        return;
      }

      const results = arr.filter(c => (c.title || '').toLowerCase().includes(q));
      if(!results.length){
        alert('No results found for "' + q + '".');
      }else{
        alert('Found ' + results.length + ' item(s) with "' + q + '". Check your folders.');
      }
    }

    function unlockContacts(){
      lockItems();

      const pwdStored = getPassword();
      if(!pwdStored){
        alert('Password is not set yet. Owner must set the password from Personal info.');
        return;
      }

      const pwd = prompt('Enter password to view contacts:');
      if(pwd === null) return;
      if(pwd === pwdStored){
        contactsUnlocked = true;
        const list = document.getElementById('contactList');
        list.style.display = 'block';
        document.getElementById('unlockBtn').textContent = '🔓 Contacts unlocked';
        renderContacts();
      } else {
        alert('Wrong password.');
      }
    }

    function saveContact(e){
      e.preventDefault();
      const name = document.getElementById('contactName').value.trim();
      const phone = document.getElementById('contactPhone').value.trim();
      const msg = document.getElementById('contactMsg').value.trim();

      if(!phone){
        alert('Please enter a phone number.');
        return;
      }

      let contacts = JSON.parse(localStorage.getItem(CONTACTS_KEY) || '[]');
      contacts.push({name, phone, msg, created:new Date().toISOString()});
      localStorage.setItem(CONTACTS_KEY, JSON.stringify(contacts));

      if(contactsUnlocked){
        renderContacts();
      }

      e.target.reset();
      alert('Contact saved safely (hidden until password is entered).');
    }

    function renderContacts(){
      const contacts = JSON.parse(localStorage.getItem(CONTACTS_KEY) || '[]');
      const box = document.getElementById('contactList');
      if(!contacts.length){
        box.textContent = 'No contacts saved yet.';
        return;
      }
      box.innerHTML = contacts.map(c =>
        `<div style="margin-bottom:6px;">
          <strong>${c.name || 'No name'}</strong> — ${c.phone}
          <div class="muted-note">${c.msg || ''}</div>
        </div>`
      ).join('');
    }

    function clearAllContacts(){

      localStorage.removeItem(CONTACTS_KEY);
      if(contactsUnlocked){
        renderContacts();
      }
    }

    function addNewNote(){
      const select = document.getElementById('noteTypeSelect');
      const type = select ? select.value : 'daily';

      if(type === 'personal'){
        const pwdStored = getPassword();
        if(!pwdStored){
          alert('Password is not set yet. Set the main password from Personal info to use personal notes.');
          return;
        }
      }

      const title = prompt('Note title:','');
      if(title === null) return;
      const content = prompt('Note content:','');
      if(content === null) return;

      const note = {
        title: title.trim(),
        content: content.trim(),
        created: new Date().toISOString()
      };
      const key = type === 'personal' ? PERSONAL_NOTES_KEY : DAILY_NOTES_KEY;
      let arr = JSON.parse(localStorage.getItem(key) || '[]');
      arr.unshift(note);
      localStorage.setItem(key, JSON.stringify(arr));
      renderNotes();
    }

    function deleteNote(type, index){
      const key = type === 'personal' ? PERSONAL_NOTES_KEY : DAILY_NOTES_KEY;
      let arr = JSON.parse(localStorage.getItem(key) || '[]');
      arr.splice(index, 1);
      localStorage.setItem(key, JSON.stringify(arr));
      renderNotes();
    }

    function renderNotes(){
      const listEl = document.getElementById('notesList');
      if(!listEl) return;
      const select = document.getElementById('noteTypeSelect');
      const type = select ? select.value : 'daily';
      const key = type === 'personal' ? PERSONAL_NOTES_KEY : DAILY_NOTES_KEY;
      const arr = JSON.parse(localStorage.getItem(key) || '[]');

      if(type === 'personal'){
        const pwdStored = getPassword();
        if(!pwdStored){
          listEl.innerHTML = '<div class="muted-note">Set main password first in Personal info to use personal notes.</div>';
          return;
        }

        if(!arr.length){
          listEl.innerHTML = '<div class="muted-note">No personal notes yet. Click + to add one. Password will be needed next time to view them.</div>';
          return;
        }

        if(!personalNotesUnlocked){
          const pwd = prompt('Enter password to view personal notes:');
          if(pwd === null || pwd !== pwdStored){
            listEl.innerHTML = '<div class="muted-note">Personal notes are locked. Wrong or no password.</div>';
            return;
          }
          personalNotesUnlocked = true;
        }
      }

      if(!arr.length && type === 'daily'){
        listEl.innerHTML =
          '<div class="muted-note">No daily notes yet. Click + to add one.</div>';
        return;
      }

      listEl.innerHTML = '';
      arr.forEach((n, idx) => {
        const div = document.createElement('div');
        div.className = 'note-card';

        const safeContent = (n.content || '').replace(/</g, '&lt;');

        div.innerHTML = `
          <div class="note-title">${n.title || 'Untitled note'}</div>
          <div class="note-body">${safeContent}</div>
          <div class="note-meta muted-note">
            ${n.created ? new Date(n.created).toLocaleString() : ''}
          </div>
        `;
        const delBtn = document.createElement('button');
        delBtn.className = 'btn';
        delBtn.textContent = 'Delete';
        delBtn.onclick = () => deleteNote(type, idx);
        div.appendChild(delBtn);

        listEl.appendChild(div);
      });
    }

    function handleNoteTypeChange(){
      const select = document.getElementById('noteTypeSelect');
      const type = select.value;
      if(type === 'personal'){
        const pwdStored = getPassword();
        if(!pwdStored){
          alert('Password is not set yet. Set the main password from Personal info to use personal notes.');
          select.value = 'daily';
          renderNotes();
          return;
        }
        personalNotesUnlocked = false;
      }
      renderNotes();
    }
